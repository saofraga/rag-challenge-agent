Evidência de reprodução — ajuste de chunking na indexação (antes/depois)
==========================================================================

Contexto: uma versão inicial do chunking (chunk_size=220, overlap=40, sem
prefixo de título por trecho) causava confusão de retrieval entre
documentos com vocabulário parecido. Este arquivo reproduz o "antes" e o
"depois" lado a lado, com comandos e output reais.

Comando de reprodução (executado diretamente, output real colado abaixo)
--------------------------------------------------------------------------

O script abaixo (i) reimplementa inline a versão anterior do chunking
(chunk_size=220, overlap=40, sem prefixo de título), sem tocar em nenhum
arquivo do repositório, construindo um índice FAISS temporário só para
esta reprodução, e (ii) carrega o índice atual do repositório
(index/bimbam_buy.faiss + index/bimbam_buy_chunks.json) para comparar
contra a mesma pergunta.

    python3 -c "
    import json
    from pathlib import Path
    import faiss
    import numpy as np
    from rag.pdf_text import extract_text
    from rag.embeddings import embed_batch, embed_query

    def chunk_text_versao_anterior(text, chunk_size=220, overlap=40):
        words = text.split()
        if not words:
            return []
        step = chunk_size - overlap
        chunks = []
        start = 0
        while start < len(words):
            chunks.append(' '.join(words[start:start + chunk_size]))
            if start + chunk_size >= len(words):
                break
            start += step
        return chunks

    SOURCE_DIR = Path('docs/fontes/bimbam-buy')
    chunks_v1 = []
    for pdf_path in sorted(SOURCE_DIR.glob('*.pdf')):
        text = extract_text(pdf_path)
        for c in chunk_text_versao_anterior(text):
            chunks_v1.append({'source': pdf_path.name, 'text': c})

    vectors_v1 = np.array(embed_batch([c['text'] for c in chunks_v1]), dtype='float32')
    faiss.normalize_L2(vectors_v1)
    index_v1 = faiss.IndexFlatIP(vectors_v1.shape[1])
    index_v1.add(vectors_v1)

    q = 'Quanto tempo demora o envio de um pedido?'

    print('QUERY:', q)
    print()
    print('--- ANTES (chunk_size=220, overlap=40, sem prefixo de titulo) ---')
    vec = np.array([embed_query(q)], dtype='float32')
    D, I = index_v1.search(vec, 3)
    for score, idx in zip(D[0], I[0]):
        print(f'  score={score:.4f}  fonte={chunks_v1[idx][\"source\"]}')
    print()

    chunks_v2 = json.load(open('index/bimbam_buy_chunks.json'))
    index_v2 = faiss.read_index('index/bimbam_buy.faiss')
    print('--- DEPOIS (indice atual do repositorio) ---')
    D2, I2 = index_v2.search(vec, 3)
    for score, idx in zip(D2[0], I2[0]):
        c = chunks_v2[idx]
        print(f'  score={score:.4f}  fonte={c[\"source\"]}  chunk_index={c[\"chunk_index\"]}')
    "
